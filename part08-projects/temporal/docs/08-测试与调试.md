# 测试与调试

Temporal 提供完善的测试和调试工具，帮助开发者确保工作流的正确性和可靠性。本文档介绍工作流单元测试、Activity Mock、时间模拟以及 Temporal Web UI 和 CLI 的使用。

---

## Workflow 单元测试

Temporal Go SDK 提供 `TestWorkflowEnvironment` 用于在测试环境中模拟工作流执行。

### TestWorkflowEnvironment 基本使用

```go
package workflow_test

import (
    "testing"
    "time"
    "go.temporal.io/sdk/testsuite"
    "go.temporal.io/sdk/workflow"
    "myapp/workflow"
)

func TestOrderWorkflow(t *testing.T) {
    // 创建测试套件
    testSuite := &testsuite.WorkflowTestSuite{}
    
    // 创建测试环境
    env := testSuite.NewTestWorkflowEnvironment()
    
    // 注册工作流
    env.RegisterWorkflow(workflow.OrderProcessingWorkflow)
    
    // 注册活动
    env.RegisterActivity(workflow.ValidateOrder)
    env.RegisterActivity(workflow.ProcessPayment)
    env.RegisterActivity(workflow.ProcessOrder)
    
    // 执行工作流
    env.ExecuteWorkflow(workflow.OrderProcessingWorkflow, &workflow.Order{
        ID:     "order-123",
        Items:  []workflow.Item{{ProductID: "prod-1", Quantity: 2}},
        Total:  100.0,
    })
    
    // 验证工作流完成
    env.AssertExpectations(t)
    
    // 获取工作流结果
    var result workflow.OrderResult
    err := env.GetWorkflowResult(&result)
    if err != nil {
        t.Fatalf("Workflow failed: %v", err)
    }
    
    // 验证结果
    if result.Status != "completed" {
        t.Errorf("Expected status 'completed', got '%s'", result.Status)
    }
}
```

### 使用 testify 断言

```go
package workflow_test

import (
    "testing"
    "github.com/stretchr/testify/assert"
    "github.com/stretchr/testify/require"

    env.RegisterActivity(workflow.ValidateOrder)
    env.RegisterActivity(workflow.ProcessPayment)

    require.NoError(t, env.GetWorkflowResult(&result))

    assert.NotEmpty(t, result.OrderID)
}
```

## Activity Mock

在测试中可以使用 Mock Activity 替代真实的 Activity 实现，控制返回值和行为。

### 基本 Mock

```go
package workflow_test

import (
    "testing"
    "go.temporal.io/sdk/testsuite"
    "myapp/workflow"
)

func TestWorkflowWithMockActivity(t *testing.T) {
    testSuite := &testsuite.WorkflowTestSuite{}
    env := testSuite.NewTestWorkflowEnvironment()
    
    env.RegisterWorkflow(workflow.PaymentWorkflow)
    
    // Mock ValidateOrder Activity
    env.OnActivity(workflow.ValidateOrder, workflow.Order{ID: "test-123"}).Return(nil)
    
    // Mock ProcessPayment Activity - 返回支付 ID
    env.OnActivity(workflow.ProcessPayment, workflow.Order{ID: "test-123"}).Return("payment-456", nil)
    
    // Mock ProcessOrder Activity
    env.OnActivity(workflow.ProcessOrder, "payment-456").Return(nil)
    
    env.ExecuteWorkflow(workflow.PaymentWorkflow, workflow.Order{ID: "test-123"})
    
    env.AssertExpectations(t)
    
    var result string
    err := env.GetWorkflowResult(&result)
    require.NoError(t, err)
    assert.Equal(t, "payment-456", result)
}
```

### Mock 动态行为

```go
func TestWorkflowWithDynamicMock(t *testing.T) {
    testSuite := &testsuite.WorkflowTestSuite{}
    env := testSuite.NewTestWorkflowEnvironment()
    
    env.RegisterWorkflow(workflow.RetryWorkflow)
    
    // Mock 动态返回值：前两次失败，第三次成功
    callCount := 0
    env.OnActivity(workflow.UnstableActivity, mock.Anything).Run(
        func(args mock.Arguments) {
            callCount++
            if callCount <= 2 {
                // 返回错误
                args.Return(errors.New("temporary failure"))
            } else {
                // 返回成功
                args.Return("success", nil)
            }
        },
    )
    
    env.ExecuteWorkflow(workflow.RetryWorkflow, "input")
    
    env.AssertExpectations(t)
    
    // 验证调用次数（包括重试）
    assert.Equal(t, 3, callCount)
}
```

## 时间模拟

测试环境中可以控制时间流逝，测试长时间运行的工作流或定时器行为。

### 基本时间控制

```go
func TestWorkflowTimer(t *testing.T) {
    testSuite := &testsuite.WorkflowTestSuite{}
    env := testSuite.NewTestWorkflowEnvironment()
    
    env.RegisterWorkflow(workflow.TimerWorkflow)
    env.RegisterActivity(workflow.DelayedActivity)
    
    // 设置起始时间
    env.SetStartTime(time.Date(2024, 1, 1, 0, 0, 0, 0, time.UTC))
    
    env.ExecuteWorkflow(workflow.TimerWorkflow, 24*time.Hour)
    
    // 工作流启动后立即检查（时间为起始时间）
    assert.False(t, env.IsWorkflowCompleted())
    
    // 模拟时间流逝
    env.Sleep(24*time.Hour)
    
    // 现在工作流应该完成
    assert.True(t, env.IsWorkflowCompleted())
    
    var result string
    err := env.GetWorkflowResult(&result)
    require.NoError(t, err)
    assert.Equal(t, "delayed-activity-completed", result)
}
```

## 表驱动测试

### 使用表驱动测试模式

```go
func TestOrderWorkflowTableDriven(t *testing.T) {
    tests := []struct {
        name          string
        order         workflow.Order
        mockSetup     func(*testsuite.TestWorkflowEnvironment)
        expectedResult workflow.OrderResult
        expectError    bool
    }{
        {
            name: "successful_order",
            order: workflow.Order{ID: "valid-123", Total: 100.0},
            mockSetup: func(env *testsuite.TestWorkflowEnvironment) {
                env.OnActivity(workflow.ValidateOrder, mock.Anything).Return(nil)
                env.OnActivity(workflow.ProcessPayment, mock.Anything).Return("pay-456", nil)
                env.OnActivity(workflow.ProcessOrder, mock.Anything).Return(nil)
            },
            expectedResult: workflow.OrderResult{Status: "completed"},
            expectError: false,
        },
        {
            name: "invalid_order",
            order: workflow.Order{ID: "invalid", Total: -1.0},
            mockSetup: func(env *testsuite.TestWorkflowEnvironment) {
                env.OnActivity(workflow.ValidateOrder, mock.Anything).Return(
                    errors.New("invalid total"),
                )
            },
            expectedResult: workflow.OrderResult{},
            expectError: true,
        },
        {
            name: "payment_failed",
            order: workflow.Order{ID: "pay-failed", Total: 100.0},
            mockSetup: func(env *testsuite.TestWorkflowEnvironment) {
                env.OnActivity(workflow.ValidateOrder, mock.Anything).Return(nil)
                env.OnActivity(workflow.ProcessPayment, mock.Anything).Return(
                    temporal.NewApplicationError("insufficient funds", "PaymentFailed", true),
                )
            },
            expectedResult: workflow.OrderResult{},
            expectError: true,
        },
    }
    
    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            testSuite := &testsuite.WorkflowTestSuite{}
            env := testSuite.NewTestWorkflowEnvironment()
            
            env.RegisterWorkflow(workflow.OrderProcessingWorkflow)
            env.RegisterActivity(workflow.ValidateOrder)
            env.RegisterActivity(workflow.ProcessPayment)
            env.RegisterActivity(workflow.ProcessOrder)
            
            tt.mockSetup(env)
            
            env.ExecuteWorkflow(workflow.OrderProcessingWorkflow, tt.order)
            
            env.AssertExpectations(t)
            
            var result workflow.OrderResult
            err := env.GetWorkflowResult(&result)
            
            if tt.expectError {
                assert.Error(t, err)
            } else {
                require.NoError(t, err)
                assert.Equal(t, tt.expectedResult.Status, result.Status)
            }
        })
    }
}
```

## CLI 调试命令

Temporal CLI 提供强大的命令行工具用于调试。

### 基本命令

```bash
# 查看命名空间列表
temporal namespace list

# 创建命名空间
temporal namespace create my-namespace --retention 7d

# 查看工作流列表
temporal workflow list

# 按条件过滤
temporal workflow list --query 'WorkflowType="OrderWorkflow" AND Status="Running"'

# 查看工作流详情
temporal workflow describe --workflow-id order-123

# 查看工作流历史
temporal workflow show --workflow-id order-123

# 查看事件历史（详细）
temporal workflow show --workflow-id order-123 --follow
```

### 工作流操作

```bash
# 启动工作流
temporal workflow start \
    --type OrderWorkflow \
    --task-queue my-queue \
    --workflow-id test-123 \
    --input '{"orderID":"123"}'

# 取消工作流
temporal workflow cancel --workflow-id order-123

# 终止工作流
temporal workflow terminate --workflow-id order-123 --reason "测试终止"

# 发送信号
temporal workflow signal \
    --workflow-id order-123 \
    --name payment-received \
    --input '{"amount":100}'

# 查询工作流
temporal workflow query \
    --workflow-id order-123 \
    --name get-status
```

### Activity 操作

```bash
# 查看任务队列信息
temporal task-queue describe --task-queue my-queue

# 重试失败的活动
temporal activity complete \
    --workflow-id order-123 \
    --activity-id 2 \
    --result '{"status":"success"}'

# 取消活动
temporal activity fail \
    --workflow-id order-123 \
    --activity-id 2 \
    --reason "手动取消"
```

### 调试命令

```bash
# 查看搜索属性
temporal workflow describe --workflow-id order-123 --json | jq '.searchAttributes'

# 查看 Memo
temporal workflow describe --workflow-id order-123 --json | jq '.memo'

# 查看事件历史（JSON 格式）
temporal workflow show --workflow-id order-123 --json

# 统计工作流数量
temporal workflow list --query 'WorkflowType="OrderWorkflow"' --json | jq 'length'

# 查看 Worker 信息
temporal task-queue describe --task-queue my-queue --json
```

```go
func DebugWorkflow(ctx workflow.Context, input string) error {
    logger := workflow.GetLogger(ctx)
    
    // 记录关键步骤
    logger.Info("Workflow started", "input", input)
    
    ao := workflow.ActivityOptions{
        StartToCloseTimeout: time.Minute,
    }
    ctx = workflow.WithActivityOptions(ctx, ao)
    
    var result string
    err := workflow.ExecuteActivity(ctx, SomeActivity, input).Get(ctx, &result)
    if err != nil {
        logger.Error("Activity failed", "error", err, "input", input)
        return err
    }
    
    logger.Info("Activity completed", "result", result)
    
    // 记录中间状态
    logger.Info("Processing result", "status", "in-progress")
    
    return nil
}
```

### 2. 使用 Query 查看状态

```go
func QueryableWorkflow(ctx workflow.Context, taskID string) error {
    var state struct {
        TaskID    string
        Status    string
        Progress  int
        StartTime time.Time
        Errors    []string
    }
    
    state.TaskID = taskID
    state.StartTime = workflow.Now(ctx)
    state.Status = "running"
    
    // 设置 Query Handler
    workflow.SetQueryHandler(ctx, "get-state", func() (interface{}, error) {
        return state, nil
    })
    
    // 工作流逻辑中更新状态
    state.Progress = 10
    
    ao := workflow.ActivityOptions{
        StartToCloseTimeout: 5 * time.Minute,
    }
    ctx = workflow.WithActivityOptions(ctx, ao)
    
    err := workflow.ExecuteActivity(ctx, SomeActivity, taskID).Get(ctx, nil)
    if err != nil {
        state.Errors = append(state.Errors, err.Error())
        state.Status = "error"
        return err
    }
    
    state.Progress = 100
    state.Status = "completed"
    
    return nil
}
```
