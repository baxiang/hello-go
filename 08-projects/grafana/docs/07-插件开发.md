# 插件开发

Grafana 提供强大的插件系统，支持开发自定义面板、数据源和应用插件。

## 插件类型

### 插件分类

| 类型 | 说明 | 用途 |
|------|------|------|
| Panel | 面板插件 | 自定义可视化 |
| DataSource | 数据源插件 | 新数据源集成 |
| App | 应用插件 | 完整应用集成 |
| Transformation | 转换插件 | 数据处理 |

## 面板插件开发

### 基础面板组件

```typescript
import React, { PureComponent } from 'react';
import { PanelProps } from '@grafana/data';
import { SimpleOptions } from './types';

interface Props extends PanelProps<SimpleOptions> {}

export class SimplePanel extends PureComponent<Props> {
  render() {
    const { data, width, height, options } = this.props;
    
    return (
      <div style={{ width, height }}>
        <h3>{options.title}</h3>
        <p>Value: {data.series[0]?.fields[0]?.values.get(0)}</p>
      </div>
    );
  }
}
```

### 配置选项

```typescript
export interface SimpleOptions {

  showValue: boolean;
  color: string;
}

export const defaults: SimpleOptions = {
  title: 'Simple Panel',
  showValue: true,
  color: '#00ff00',
};
```

### 配置编辑器

```typescript
import React, { PureComponent } from 'react';
import { PanelOptionsEditorProps } from '@grafana/data';
import { SimpleOptions } from './types';

export class SimpleEditor extends PureComponent<PanelOptionsEditorProps<SimpleOptions>> {
  onTitleChange = (event: React.ChangeEvent<HTMLInputElement>) => {
    this.props.onOptionsChange({ ...this.props.options, title: event.target.value });
  };

  onColorChange = (color: string) => {
    this.props.onOptionsChange({ ...this.props.options, color: color });
  };

  render() {
    const { options } = this.props;
    
    return (
      <div className="editor-section">
        <div className="editor-row">
          <label>Title</label>
          <input
            type="text"
            value={options.title}
            onChange={this.onTitleChange}
          />
        </div>
        <div className="editor-row">
          <label>Color</label>
          <input
            type="color"
            value={options.color}
            onChange={(e) => this.onColorChange(e.target.value)}
          />
        </div>
      </div>
    );
  }
}
```

### 模块注册

```typescript
import { PanelPlugin } from '@grafana/data';
import { SimplePanel } from './components/SimplePanel';
import { SimpleEditor } from './components/SimpleEditor';
import { defaults } from './types';

export const plugin = new PanelPlugin<SimpleOptions>(SimplePanel)
  .setDefaults(defaults)
  .setEditor(SimpleEditor);
```

## 数据源插件开发

### 数据源组件

```typescript
import { DataSourceApi, DataSourceInstanceSettings } from '@grafana/data';
import { MyDataSourceOptions, MyQuery } from './types';

export class MyDataSource extends DataSourceApi<MyQuery, MyDataSourceOptions> {
  constructor(instanceSettings: DataSourceInstanceSettings<MyDataSourceOptions>) {
    super(instanceSettings);
    this.url = instanceSettings.url;
  }

  async query(options: DataQueryRequest<MyQuery>): Promise<DataQueryResponse> {
    const { targets } = options;
    const data = [];

    for (const target of targets) {
      const response = await this.doRequest(target);
      data.push(response);
    }

    return { data };
  }

  async doRequest(query: MyQuery): Promise<any> {
    const response = await fetch(`${this.url}/api/query`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(query),
    });
    
    return response.json();
  }

  async testDatasource(): Promise<any> {
    try {
      const response = await fetch(`${this.url}/health`);
      if (response.ok) {
        return { status: 'success', message: 'Data source is working' };
      }
      return { status: 'error', message: 'Data source is not working' };
    } catch (error) {
      return { status: 'error', message: error.message };
    }
  }
}
```

### 配置编辑器

```typescript
import React, { PureComponent } from 'react';
import { DataSourceSettings } from '@grafana/data';
import { MyDataSourceOptions } from './types';

interface Props {
  options: DataSourceSettings<MyDataSourceOptions>;
  onOptionsChange: (options: DataSourceSettings<MyDataSourceOptions>) => void;
}

export class ConfigEditor extends PureComponent<Props> {
  onUrlChange = (event: React.ChangeEvent<HTMLInputElement>) => {
    const { options, onOptionsChange } = this.props;
    onOptionsChange({
      ...options,
      jsonData: {
        ...options.jsonData,
        url: event.target.value,
      },
    });
  };

  render() {
    const { options } = this.props;
    
    return (
      <div className="gf-form-group">
        <div className="gf-form">
          <label className="gf-form-label">URL</label>
          <input
            className="gf-form-input"
            type="text"
            value={options.jsonData.url || ''}
            onChange={this.onUrlChange}
            placeholder="http://localhost:8080"
          />
        </div>
      </div>
    );
  }
}
```

### 查询编辑器

```typescript
import React, { PureComponent } from 'react';
import { QueryEditorProps } from '@grafana/data';
import { MyDataSource } from './datasource';
import { MyQuery, MyDataSourceOptions } from './types';

type Props = QueryEditorProps<MyDataSource, MyQuery, MyDataSourceOptions>;

export class QueryEditor extends PureComponent<Props> {
  onQueryTextChange = (event: React.ChangeEvent<HTMLInputElement>) => {
    const { query, onChange, onRunQuery } = this.props;
    onChange({ ...query, queryText: event.target.value });
    onRunQuery();
  };

  render() {
    const { query } = this.props;
    
    return (
      <div className="gf-form">
        <label className="gf-form-label">Query</label>
        <input
          className="gf-form-input"
          type="text"
          value={query.queryText || ''}
          onChange={this.onQueryTextChange}
          placeholder="Enter query"
        />
      </div>
    );
  }
}
```

### 模块注册

```typescript
import { DataSourcePlugin } from '@grafana/data';
```

## 开发环境设置

### 初始化项目

```bash

npx @grafana/create-plugin@latest

# 或手动创建
mkdir my-plugin && cd my-plugin
npm init -y
npm install @grafana/data @grafana/ui @grafana/runtime
npm install --save-dev webpack webpack-cli typescript ts-loader
```
